async function seedFacultyAccount(params: {
  facultyCode: string;
  fullName: string;
  email: string;
  department: string;
  passwordHash: string;
  reset: boolean;
}) {
  const faculty = await prisma.faculty.upsert({
    where: { facultyCode: params.facultyCode },
    update: {
      fullName: params.fullName,
      email: params.email,
      department: params.department,
      passwordHash: params.passwordHash,
    },
    create: {
      facultyCode: params.facultyCode,
      fullName: params.fullName,
      email: params.email,
      department: params.department,
      passwordHash: params.passwordHash,
    },
  });

  const existingCount = await prisma.event.count({
    where: { facultyId: faculty.id },
  });

  if (existingCount > 0 && !params.reset) {
    console.log(
      `Skipping event creation for ${params.facultyCode} (${existingCount} events already exist). Use --reset to recreate.`
    );
    return;
  }

  if (params.reset) {
    await prisma.event.deleteMany({ where: { facultyId: faculty.id } });
  }

  const now = startOfHour(new Date());
  const eventsToCreate = [];

  for (let i = 0; i < 48; i++) {
    const template = EVENT_TEMPLATES[i % EVENT_TEMPLATES.length];
    const daysAgo = i * 7 + (i % 5);
    const startAt = subDays(now, daysAgo);
    const endAt = addHours(startAt, 2);

    eventsToCreate.push({
      facultyId: faculty.id,
      title:
        i === 3
          ? "Applied AI Workshop"
          : `${template.title} #${Math.floor(i / 8) + 1}`,
      type: template.type,
      startAt,
      endAt,
      allDay: template.type === EventType.BUSY && i % 2 === 0,
      location: LOCATIONS[i % LOCATIONS.length],
      department: DEPARTMENTS[i % DEPARTMENTS.length],
      organizer: template.organizer,
      role: template.role,
      description: template.description,
    });
  }

  for (let u = 1; u <= 4; u++) {
    const template = EVENT_TEMPLATES[u % EVENT_TEMPLATES.length];
    const startAt = addDays(now, u * 4);
    const endAt = addHours(startAt, 2);

    eventsToCreate.push({
      facultyId: faculty.id,
      title: `Upcoming: ${template.title}`,
      type: template.type,
      startAt,
      endAt,
      allDay: false,
      location: LOCATIONS[u % LOCATIONS.length],
      department: DEPARTMENTS[u % DEPARTMENTS.length],
      organizer: template.organizer,
      role: template.role,
      description: template.description,
    });
  }

  await prisma.event.createMany({ data: eventsToCreate });
  console.log(
    `Seeded ${eventsToCreate.length} events for ${params.facultyCode}.`
  );
}

export function getCalendarDaysBetween(startAt: Date, endAt: Date): string[] {
  const tz = APP_CONFIG.timezone;
  const zonedStart = startOfDay(toZonedTime(startAt, tz));
  const zonedEnd = endOfDay(toZonedTime(endAt, tz));
  if (zonedEnd < zonedStart) return [format(zonedStart, "yyyy-MM-dd")];

  return eachDayOfInterval({ start: zonedStart, end: zonedEnd }).map((d) =>
    format(d, "yyyy-MM-dd")
  );
}

export function getAcademicYearRange(referenceDate: Date = new Date()): {
  from: Date;
  to: Date;
} {
  const tz = APP_CONFIG.timezone;
  const zonedNow = toZonedTime(referenceDate, tz);
  const startMonthIdx = APP_CONFIG.academicYearStartMonth - 1; // 0-indexed (July = 6)
  const currentYear = zonedNow.getFullYear();
  const currentMonth = zonedNow.getMonth();

  const startYear = currentMonth >= startMonthIdx ? currentYear : currentYear - 1;
  const startLocal = new Date(startYear, startMonthIdx, 1, 0, 0, 0, 0);
  const endLocal = endOfMonth(new Date(startYear + 1, startMonthIdx - 1, 1));

  return {
    from: fromZonedTime(startLocal, tz),
    to: fromZonedTime(endLocal, tz),
  };
}

export function getCurrentSemesterRange(referenceDate: Date = new Date()): {
  from: Date;
  to: Date;
} {
  const tz = APP_CONFIG.timezone;
  const zonedNow = toZonedTime(referenceDate, tz);
  const startMonthIdx = APP_CONFIG.academicYearStartMonth - 1; // e.g., 6 for July
  const secondSemMonthIdx = (startMonthIdx + 6) % 12; // e.g., 0 for Jan
  const year = zonedNow.getFullYear();
  const month = zonedNow.getMonth();

  let semStartLocal: Date;
  let semEndLocal: Date;

  if (startMonthIdx < secondSemMonthIdx) {
    if (month >= startMonthIdx && month < secondSemMonthIdx) {
      semStartLocal = new Date(year, startMonthIdx, 1, 0, 0, 0, 0);
      semEndLocal = endOfMonth(new Date(year, secondSemMonthIdx - 1, 1));
    } else {
      const sYear = month >= secondSemMonthIdx ? year : year - 1;
      semStartLocal = new Date(sYear, secondSemMonthIdx, 1, 0, 0, 0, 0);
      semEndLocal = endOfMonth(new Date(sYear + 1, startMonthIdx - 1, 1));
    }
  } else {
    // e.g., July (6) to Dec (11) and Jan (0) to June (5)
    if (month >= startMonthIdx) {
      semStartLocal = new Date(year, startMonthIdx, 1, 0, 0, 0, 0);
      semEndLocal = endOfMonth(new Date(year, 11, 1));
    } else if (month < startMonthIdx && month >= secondSemMonthIdx) {
      semStartLocal = new Date(year, secondSemMonthIdx, 1, 0, 0, 0, 0);
      semEndLocal = endOfMonth(new Date(year, startMonthIdx - 1, 1));
    } else {
      semStartLocal = new Date(year - 1, startMonthIdx, 1, 0, 0, 0, 0);
      semEndLocal = endOfMonth(new Date(year, secondSemMonthIdx - 1, 1));
    }
  }

  return {
    from: fromZonedTime(semStartLocal, tz),
    to: fromZonedTime(semEndLocal, tz),
  };
}

export function getPresetDateRange(
  preset:
    | "this_week"
    | "this_month"
    | "last_3_months"
    | "last_6_months"
    | "academic_year",
  now: Date = new Date()
): { from: Date; to: Date } {
  const tz = APP_CONFIG.timezone;
  const zonedNow = toZonedTime(now, tz);

  switch (preset) {
    case "this_week":
      return {
        from: fromZonedTime(startOfWeek(zonedNow, { weekStartsOn: 1 }), tz),
        to: fromZonedTime(endOfWeek(zonedNow, { weekStartsOn: 1 }), tz),
      };
    case "this_month":
      return {
        from: fromZonedTime(startOfMonth(zonedNow), tz),
        to: fromZonedTime(endOfMonth(zonedNow), tz),
      };
    case "last_3_months":
      return {
        from: fromZonedTime(startOfDay(subMonths(zonedNow, 3)), tz),
        to: fromZonedTime(endOfDay(zonedNow), tz),
      };
    case "last_6_months":
      return {
        from: fromZonedTime(startOfDay(subMonths(zonedNow, 6)), tz),
        to: fromZonedTime(endOfDay(zonedNow), tz),
      };
    case "academic_year":
      return getAcademicYearRange(now);
  }
}

        code,

      },

export function handleZodError(error: ZodError) {
  const fields: Record<string, string> = {};
  for (const issue of error.issues) {
    const key = issue.path.join(".") || "root";
    if (!fields[key]) {
      fields[key] = issue.message;
    }
  }
  const firstMessage = error.issues[0]?.message || "Invalid input provided.";
  return apiError("VALIDATION_ERROR", firstMessage, 400, fields);
}

import { SignJWT, jwtVerify } from "jose";
import { APP_CONFIG } from "@/config/constants";

export interface SessionPayload {
  facultyId: string;
  facultyCode: string;
  email: string;
}

function getSecretKey(): Uint8Array {
  const secret = process.env.JWT_SECRET;
  if (!secret || secret.length < 32) {
    throw new Error("JWT_SECRET must be set and at least 32 characters long.");
  }
  return new TextEncoder().encode(secret);
}

export async function signSessionToken(payload: SessionPayload): Promise<string> {
  return new SignJWT({ ...payload })
    .setProtectedHeader({ alg: "HS256" })
    .setIssuedAt()
    .setExpirationTime(`${APP_CONFIG.sessionDays}d`)
    .sign(getSecretKey());
}

export async function verifySessionToken(
  token: string
): Promise<SessionPayload | null> {
  try {
    const { payload } = await jwtVerify(token, getSecretKey());
    if (
      typeof payload.facultyId === "string" &&
      typeof payload.facultyCode === "string" &&
      typeof payload.email === "string"
    ) {
      return {
        facultyId: payload.facultyId,
        facultyCode: payload.facultyCode,
        email: payload.email,
      };
    }
    return null;
  } catch {
    return null;
  }
}

interface RateRecord {
  count: number;
  windowStartMs: number;
  blockedUntilMs?: number;
}

const loginAttempts = new Map<string, RateRecord>();
const aiSearchAttempts = new Map<string, RateRecord>();

const FIFTEEN_MINUTES_MS = 15 * 60 * 1000;
const ONE_HOUR_MS = 60 * 60 * 1000;
const MAX_LOGIN_ATTEMPTS = 5;
const MAX_AI_REQUESTS_PER_HOUR = 30;

/**
 * F1.10: After 5 failed sign-in attempts for the same faculty ID within 15 minutes,
 * block further attempts for 15 minutes.
 */
export function checkLoginRateLimit(facultyCode: string, nowMs = Date.now()): {
  allowed: boolean;
} {
  const key = facultyCode.toUpperCase();
  const record = loginAttempts.get(key);
  if (!record) return { allowed: true };

  if (record.blockedUntilMs && nowMs < record.blockedUntilMs) {
    return { allowed: false };
  }

  if (nowMs - record.windowStartMs > FIFTEEN_MINUTES_MS) {
    loginAttempts.delete(key);
    return { allowed: true };
  }

  return { allowed: record.count < MAX_LOGIN_ATTEMPTS };
}

export function recordFailedLogin(facultyCode: string, nowMs = Date.now()): void {
  const key = facultyCode.toUpperCase();
  const existing = loginAttempts.get(key);

  if (!existing || nowMs - existing.windowStartMs > FIFTEEN_MINUTES_MS) {
    loginAttempts.set(key, { count: 1, windowStartMs: nowMs });
    return;
  }

  const nextCount = existing.count + 1;
  const blockedUntilMs =
    nextCount >= MAX_LOGIN_ATTEMPTS ? nowMs + FIFTEEN_MINUTES_MS : undefined;

  loginAttempts.set(key, {
    count: nextCount,
    windowStartMs: existing.windowStartMs,
    blockedUntilMs,
  });
}

export function clearLoginRateLimit(facultyCode: string): void {
  loginAttempts.delete(facultyCode.toUpperCase());
}

/**
 * SEC7: Rate limit on AI search (30 requests per user per hour).
 */
export function consumeAiRateLimit(facultyId: string, nowMs = Date.now()): {
  allowed: boolean;
} {
  const existing = aiSearchAttempts.get(facultyId);
  if (!existing || nowMs - existing.windowStartMs > ONE_HOUR_MS) {
    aiSearchAttempts.set(facultyId, { count: 1, windowStartMs: nowMs });
    return { allowed: true };
  }

  if (existing.count >= MAX_AI_REQUESTS_PER_HOUR) {
    return { allowed: false };
  }

  existing.count += 1;
  return { allowed: true };
}

export function setSessionCookie(response: NextResponse, token: string): void {
  response.cookies.set(APP_CONFIG.sessionCookieName, token, {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    path: "/",
    maxAge: APP_CONFIG.sessionDays * 24 * 60 * 60,
  });
}

class CredentialsAuthProvider implements AuthProvider {
  async signIn(facultyCode: string, passwordPlain: string) {
    const normalizedCode = facultyCode.trim().toUpperCase();
    const faculty = await prisma.faculty.findUnique({
      where: { facultyCode: normalizedCode },
    });

    if (!faculty) {
      await hashPassword(passwordPlain);
      return null;
    }

    const isValid = await verifyPassword(passwordPlain, faculty.passwordHash);
    if (!isValid) return null;

    const token = await signSessionToken({
      facultyId: faculty.id,
      facultyCode: faculty.facultyCode,
      email: faculty.email,
    });

    return {
      profile: toPublicProfile(faculty),
      token,
    };
  }

  async getSession(): Promise<SessionPayload | null> {
    const cookieStore = await cookies();
    const token = cookieStore.get(APP_CONFIG.sessionCookieName)?.value;
    if (!token) return null;
    return verifySessionToken(token);
  }

  signOut(response: NextResponse): void {
    response.cookies.set(APP_CONFIG.sessionCookieName, "", {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "lax",
      path: "/",
      maxAge: 0,
    });
  }
}

export const authProvider: AuthProvider = new CredentialsAuthProvider();

export interface OverlappingEventSummary {
  id: string;
  title: string;
  startAt: Date;
  endAt: Date;
}

/**
 * F2.8 & Section 15.1: Returns true for partial and full overlap, false for adjacent events.
 */
export function intervalsOverlap(
  startA: Date,
  endA: Date,
  startB: Date,
  endB: Date
): boolean {
  if (startA.getTime() === endA.getTime() && startB.getTime() === endB.getTime()) {
    return startA.getTime() === startB.getTime();
  }
  return startA < endB && endA > startB;
}

export async function findOverlappingEvents(params: {
  facultyId: string;
  startAt: Date;
  endAt: Date;
  excludeId?: string;
}): Promise<OverlappingEventSummary[]> {
  const { facultyId, startAt, endAt, excludeId } = params;

  const candidates = await prisma.event.findMany({
    where: {
      facultyId, // SEC1: Mandatory scoping
      ...(excludeId ? { id: { not: excludeId } } : {}),
      startAt: { lt: endAt },
      endAt: { gt: startAt },
    },
    select: {
      id: true,
      title: true,
      startAt: true,
      endAt: true,
    },
    orderBy: { startAt: "asc" },
  });

  return candidates.filter((c) =>
    intervalsOverlap(startAt, endAt, c.startAt, c.endAt)
  );
}

export interface EventListFilters {

    | "this_week"

    | "last_6_months"
    | "academic_year";

export function parseEventTypesParam(typeParam?: string): EventType[] {
  if (!typeParam) return [];
  const valid = new Set<string>(Object.values(EventType));
  return typeParam
    .split(",")
    .map((s) => s.trim().toUpperCase())
    .filter((s): s is EventType => valid.has(s));
}

export function normalizeAllDayDates(
  startAt: Date,
  endAt: Date,
  allDay: boolean
): { startAt: Date; endAt: Date } {
  if (!allDay) return { startAt, endAt };
  const tz = APP_CONFIG.timezone;
  const zonedStart = startOfDay(toZonedTime(startAt, tz));
  const zonedEnd = endOfDay(toZonedTime(endAt, tz));
  return {
    startAt: fromZonedTime(zonedStart, tz),
    endAt: fromZonedTime(zonedEnd, tz),
  };
}

export function buildEventWhereClause(
  facultyId: string,
  filters: Omit<EventListFilters, "page" | "pageSize">
): Prisma.EventWhereInput {
  // SEC1: Every query includes facultyId in the where clause
  const where: Prisma.EventWhereInput = { facultyId };

  let rangeFrom = filters.from;
  let rangeTo = filters.to;

  if (filters.preset && !rangeFrom && !rangeTo) {
    const presetRange = getPresetDateRange(filters.preset);
    rangeFrom = presetRange.from;
    rangeTo = presetRange.to;
  }

  if (rangeFrom || rangeTo) {
    where.startAt = {
      ...(rangeFrom ? { gte: rangeFrom } : {}),
      ...(rangeTo ? { lte: rangeTo } : {}),
    };
  }

  const parsedTypes = parseEventTypesParam(filters.type);
  if (parsedTypes.length === 1) {
    where.type = parsedTypes[0];
  } else if (parsedTypes.length > 1) {
    where.type = { in: parsedTypes };
  }

  if (filters.department) {
    where.department = {
      equals: filters.department,
      mode: "insensitive",
    };
  }

  if (filters.q) {
    const q = filters.q;
    where.OR = [
      { title: { contains: q, mode: "insensitive" } },
      { location: { contains: q, mode: "insensitive" } },
      { organizer: { contains: q, mode: "insensitive" } },
      { description: { contains: q, mode: "insensitive" } },
    ];
  }

  return where;
}

export async function listFacultyEvents(
  facultyId: string,
  filters: EventListFilters
) {
  const where = buildEventWhereClause(facultyId, filters);
  const skip = (filters.page - 1) * filters.pageSize;

  const [items, total] = await Promise.all([
    prisma.event.findMany({
      where,
      orderBy: { startAt: "desc" },
      skip,
      take: filters.pageSize,
    }),
    prisma.event.count({ where }),
  ]);

  return {
    items,
    total,
    page: filters.page,
    pageSize: filters.pageSize,
  };
}

import { Event, EventType } from "@prisma/client";
import { subMonths, startOfMonth, format } from "date-fns";
import { toZonedTime } from "date-fns-tz";
import { APP_CONFIG } from "@/config/constants";
import {
  formatMonthKeyInAppTz,
  formatYearKeyInAppTz,
  getCalendarDaysBetween,
} from "@/lib/time";
import { evaluateBadges, EvaluatedBadge } from "./badges";

export interface AchievementsResult {
  metrics: {
    totalHosted: number;
    totalAttended: number;
    totalBusyDays: number;
    totalEvents: number;
    busiestMonth: { month: string; count: number } | null;
    activeMonthsStreak: number;
    departmentsEngaged: number;
  };
  eventsPerYear: Array<{ year: string; count: number }>;
  eventsPerMonthLast12: Array<{
    month: string;
    label: string;
    ATTENDED: number;
    HOSTED: number;
    BUSY: number;
    total: number;
  }>;
  typeBreakdown: Array<{ type: EventType; count: number }>;
  badges: EvaluatedBadge[];
  topBadge: EvaluatedBadge | null;
}

function computeLongestMonthStreak(sortedMonthKeys: string[]): number {
  if (sortedMonthKeys.length === 0) return 0;
  let maxStreak = 1;
  let currentStreak = 1;

  for (let i = 1; i < sortedMonthKeys.length; i++) {
    const [prevY, prevM] = sortedMonthKeys[i - 1].split("-").map(Number);
    const [currY, currM] = sortedMonthKeys[i].split("-").map(Number);

    const prevTotalMonths = prevY * 12 + prevM;
    const currTotalMonths = currY * 12 + currM;

    if (currTotalMonths === prevTotalMonths + 1) {
      currentStreak += 1;
      if (currentStreak > maxStreak) maxStreak = currentStreak;
    } else if (currTotalMonths > prevTotalMonths + 1) {
      currentStreak = 1;
    }
  }

  return maxStreak;
}

export function calculateAchievementsFromEvents(
  events: Array<
    Pick<Event, "type" | "startAt" | "endAt" | "department">
  >,
  referenceNow: Date = new Date()
): AchievementsResult {
  let totalHosted = 0;
  let totalAttended = 0;
  let totalBusyEvents = 0;

  const busyDaySet = new Set<string>();
  const departmentsSet = new Set<string>();
  const yearCounts = new Map<string, number>();
  const monthCounts = new Map<string, number>();
  const monthsByYear = new Map<string, Set<string>>();

  for (const ev of events) {
    if (ev.type === EventType.HOSTED) totalHosted += 1;
    if (ev.type === EventType.ATTENDED) totalAttended += 1;
    if (ev.type === EventType.BUSY) {
      totalBusyEvents += 1;
      for (const dayStr of getCalendarDaysBetween(ev.startAt, ev.endAt)) {
        busyDaySet.add(dayStr);
      }
    }

    if (ev.department?.trim()) {
      departmentsSet.add(ev.department.trim().toLowerCase());
    }

    const yearKey = formatYearKeyInAppTz(ev.startAt);
    const monthKey = formatMonthKeyInAppTz(ev.startAt);

    yearCounts.set(yearKey, (yearCounts.get(yearKey) ?? 0) + 1);
    monthCounts.set(monthKey, (monthCounts.get(monthKey) ?? 0) + 1);

    if (!monthsByYear.has(yearKey)) {
      monthsByYear.set(yearKey, new Set());
    }
    monthsByYear.get(yearKey)!.add(monthKey);
  }

  let busiestMonth: { month: string; count: number } | null = null;
  for (const [month, count] of monthCounts.entries()) {
    if (
      !busiestMonth ||
      count > busiestMonth.count ||
      (count === busiestMonth.count && month > busiestMonth.month)
    ) {
      busiestMonth = { month, count };
    }
  }

  const sortedDistinctMonths = Array.from(monthCounts.keys()).sort();
  const activeMonthsStreak = computeLongestMonthStreak(sortedDistinctMonths);

  let hasFullCalendarYear = false;
  for (const monthSet of monthsByYear.values()) {
    if (monthSet.size === 12) {
      hasFullCalendarYear = true;
      break;
    }
  }

  const zonedNow = toZonedTime(referenceNow, APP_CONFIG.timezone);
  const eventsPerMonthLast12: AchievementsResult["eventsPerMonthLast12"] = [];
  const last12Map = new Map<
    string,
    {
      month: string;
      label: string;
      ATTENDED: number;
      HOSTED: number;
      BUSY: number;
      total: number;
    }
  >();

  for (let i = 11; i >= 0; i--) {
    const mDate = startOfMonth(subMonths(zonedNow, i));
    const key = format(mDate, "yyyy-MM");
    const label = format(mDate, "MMM yyyy");
    const entry = { month: key, label, ATTENDED: 0, HOSTED: 0, BUSY: 0, total: 0 };
    eventsPerMonthLast12.push(entry);
    last12Map.set(key, entry);
  }

  for (const ev of events) {
    const mKey = formatMonthKeyInAppTz(ev.startAt);
    const bucket = last12Map.get(mKey);
    if (bucket) {
      bucket[ev.type] += 1;
      bucket.total += 1;
    }
  }

  const eventsPerYear = Array.from(yearCounts.entries())
    .map(([year, count]) => ({ year, count }))
    .sort((a, b) => a.year.localeCompare(b.year));

  const { badges, topBadge } = evaluateBadges({
    totalHosted,
    totalAttended,
    distinctMonthsCount: sortedDistinctMonths.length,
    departmentsEngaged: departmentsSet.size,
    hasFullCalendarYear,
  });

  return {
    metrics: {
      totalHosted,
      totalAttended,
      totalBusyDays: busyDaySet.size,
      totalEvents: events.length,
      busiestMonth,
      activeMonthsStreak,
      departmentsEngaged: departmentsSet.size,
    },
    eventsPerYear,
    eventsPerMonthLast12,
    typeBreakdown: [
      { type: EventType.ATTENDED, count: totalAttended },
      { type: EventType.HOSTED, count: totalHosted },
      { type: EventType.BUSY, count: totalBusyEvents },
    ],
    badges,
    topBadge,
  };
}

    if (ev.type === EventType.ATTENDED) totalAttended += 1;

      id: ev.id,
      dateFormatted: formatInAppTz(
        ev.startAt,
        ev.allDay ? "dd MMM yyyy (All day)" : "dd MMM yyyy, hh:mm a"
      ),
      startAt: ev.startAt.toISOString(),
      endAt: ev.endAt.toISOString(),
      title: ev.title,
      type: ev.type,
      role: ev.role ?? "-",
      organizer: ev.organizer ?? "-",
      location: ev.location ?? "-",
      department: ev.department ?? "-",
    };
  });

  const generatedDate = formatInAppTz(new Date(), "dd MMM yyyy");

  return {
    header: {
      title: params.title || "Semester Activity Report",
      facultyName: faculty.fullName,
      facultyCode: faculty.facultyCode,
      department: faculty.department,
      from: formatInAppTz(params.from, "yyyy-MM-dd"),
      to: formatInAppTz(params.to, "yyyy-MM-dd"),
      generatedDate,
    },
    summary: {
      totalHosted,
      totalAttended,
      totalBusyDays: busyDays.size,
      totalEvents: rows.length,
      departmentsInvolved: Array.from(deptSet).sort(),
    },
    events: rows,
    footer: `Generated by FacultyPulse on ${generatedDate}. Records are self-reported by the faculty member.`,
  };
}

const TYPE_PATTERNS: Array<{ type: EventType; phrases: string[] }> = [
  {
    type: EventType.HOSTED,
    phrases: ["hosted", "host", "organized", "conducted", "chaired"],
  },
  {
    type: EventType.ATTENDED,
    phrases: ["attended", "went to", "joined", "participated"],
  },
  {
    type: EventType.BUSY,
    phrases: ["busy days", "busy day", "busy", "occupied", "meetings"],
  },
];

/**
 * Pipeline Step 2: Detect event type and return matched phrases so they can be
 * stripped from keyword extraction.
 */
export function detectEventType(normalized: string): {
  type?: EventType;
  matchedPhrases: string[];
} {
  for (const group of TYPE_PATTERNS) {
    for (const phrase of group.phrases) {
      const regex = new RegExp(`\\b${phrase.replace(/\s+/g, "\\s+")}\\b`, "i");
      if (regex.test(normalized)) {
        return { type: group.type, matchedPhrases: [phrase] };
      }
    }
  }
  return { matchedPhrases: [] };
}

/**
 * Pipeline Step 3: Extract date range (Absolute, Relative, Seasons, Semester).
 */
export function extractDateRange(
  normalized: string,
  referenceNow: Date = new Date()
): ExtractedDateResult {
  const tz = APP_CONFIG.timezone;
  const zonedNow = toZonedTime(referenceNow, tz);
  const currentYear = zonedNow.getFullYear();
  const currentMonth = zonedNow.getMonth();

  const toUtcRange = (startLocal: Date, endLocal: Date) => ({
    from: fromZonedTime(startOfDay(startLocal), tz),
    to: fromZonedTime(endOfDay(endLocal), tz),
  });

  if (/\btoday\b/.test(normalized)) {
    return {
      ...toUtcRange(zonedNow, zonedNow),
      label: "today",
      matchedPhrases: ["today"],
    };
  }

  if (/\byesterday\b/.test(normalized)) {
    const yest = subDays(zonedNow, 1);
    return {
      ...toUtcRange(yest, yest),
      label: "yesterday",
      matchedPhrases: ["yesterday"],
    };
  }

  if (/\bthis week\b/.test(normalized)) {
    return {
      ...toUtcRange(
        startOfWeek(zonedNow, { weekStartsOn: 1 }),
        endOfWeek(zonedNow, { weekStartsOn: 1 })
      ),
      label: "this week",
      matchedPhrases: ["this week"],
    };
  }

  if (/\blast week\b/.test(normalized)) {
    const lw = subWeeks(zonedNow, 1);
    return {
      ...toUtcRange(
        startOfWeek(lw, { weekStartsOn: 1 }),
        endOfWeek(lw, { weekStartsOn: 1 })
      ),
      label: "last week",
      matchedPhrases: ["last week"],
    };
  }

  const lastNMonthsMatch = normalized.match(/\blast\s+(\d+)\s+months?\b/);
  if (lastNMonthsMatch) {
    const n = Number(lastNMonthsMatch[1]);
    return {
      ...toUtcRange(subMonths(zonedNow, n), zonedNow),
      label: lastNMonthsMatch[0],
      matchedPhrases: [lastNMonthsMatch[0]],
    };
  }

  if (/\bthis month\b/.test(normalized)) {
    return {
      ...toUtcRange(startOfMonth(zonedNow), endOfMonth(zonedNow)),
      label: "this month",
      matchedPhrases: ["this month"],
    };
  }

  if (/\blast month\b/.test(normalized)) {
    const lm = subMonths(zonedNow, 1);
    return {
      ...toUtcRange(startOfMonth(lm), endOfMonth(lm)),
      label: "last month",
      matchedPhrases: ["last month"],
    };
  }

  if (/\bthis year\b/.test(normalized)) {
    return {
      ...toUtcRange(startOfYear(zonedNow), endOfYear(zonedNow)),
      label: "this year",
      matchedPhrases: ["this year"],
    };
  }

  if (/\blast year\b/.test(normalized)) {
    const ly = subYears(zonedNow, 1);
    return {
      ...toUtcRange(startOfYear(ly), endOfYear(ly)),
      label: "last year",
      matchedPhrases: ["last year"],
    };
  }

  if (/\bthis semester\b/.test(normalized)) {
    const sem = getCurrentSemesterRange(referenceNow);
    return {
      from: sem.from,
      to: sem.to,
      label: "this semester",
      matchedPhrases: ["this semester"],
    };
  }

  if (/\blast semester\b/.test(normalized)) {
    const currentSem = getCurrentSemesterRange(referenceNow);
    const prevRef = subDays(currentSem.from, 10);
    const prevSem = getCurrentSemesterRange(prevRef);
    return {
      from: prevSem.from,
      to: prevSem.to,
      label: "last semester",
      matchedPhrases: ["last semester"],
    };
  }

  const seasonMatch = normalized.match(
    /\b(?:(last|this)\s+)?(spring|summer|autumn|fall|winter)(?:\s+(20\d{2}))?\b/
  );
  if (seasonMatch) {
    const modifier = seasonMatch[1];
    const season = seasonMatch[2];
    const explicitYear = seasonMatch[3] ? Number(seasonMatch[3]) : undefined;

    const seasonMonths: Record<string, [number, number]> = {
      spring: [2, 4],
      summer: [5, 7],
      autumn: [8, 10],
      fall: [8, 10],
      winter: [11, 1],
    };

    const [startM, endM] = seasonMonths[season];
    let targetYear = explicitYear ?? currentYear;

    if (!explicitYear) {
      if (modifier === "last") {
        targetYear = currentYear - 1;
      } else if (modifier !== "this") {
        if (currentMonth <= endM) {
          targetYear = currentYear - 1;
        }
      }
    }

    if (season === "winter") {
      const startLocal = new Date(targetYear, 11, 1);
      const endLocal = endOfMonth(new Date(targetYear + 1, 1, 1));
      return {
        ...toUtcRange(startLocal, endLocal),
        label: seasonMatch[0],
        matchedPhrases: [seasonMatch[0]],
      };
    } else {
      const startLocal = new Date(targetYear, startM, 1);
      const endLocal = endOfMonth(new Date(targetYear, endM, 1));
      return {
        ...toUtcRange(startLocal, endLocal),
        label: seasonMatch[0],
        matchedPhrases: [seasonMatch[0]],
      };
    }
  }

  const monthNamesPattern = Object.keys(MONTH_MAP).join("|");
  const dmyRegex = new RegExp(
    `\\b(\\d{1,2})\\s+(${monthNamesPattern})\\s+(20\\d{2})\\b`
  );
  const dmyMatch = normalized.match(dmyRegex);
  if (dmyMatch) {
    const day = Number(dmyMatch[1]);
    const monthIdx = MONTH_MAP[dmyMatch[2]];
    const year = Number(dmyMatch[3]);
    const specificDate = new Date(year, monthIdx, day);
    return {
      ...toUtcRange(specificDate, specificDate),
      label: dmyMatch[0],
      matchedPhrases: [dmyMatch[0]],
    };
  }

  const myRegex = new RegExp(`\\b(${monthNamesPattern})\\s+(20\\d{2})\\b`);
  const myMatch = normalized.match(myRegex);
  if (myMatch) {
    const monthIdx = MONTH_MAP[myMatch[1]];
    const year = Number(myMatch[2]);
    const mStart = new Date(year, monthIdx, 1);
    const mEnd = endOfMonth(mStart);
    return {
      ...toUtcRange(mStart, mEnd),
      label: myMatch[0],
      matchedPhrases: [myMatch[0]],
    };
  }

  const yearMatch = normalized.match(/\b(20\d{2})\b/);
  if (yearMatch) {
    const year = Number(yearMatch[1]);
    const yStart = new Date(year, 0, 1);
    const yEnd = new Date(year, 11, 31);
    return {
      ...toUtcRange(yStart, yEnd),
      label: yearMatch[1],
      matchedPhrases: [yearMatch[0]],
    };
  }

  return { matchedPhrases: [] };
}

function bigrams(str: string): Set<string> {
  const s = str.toLowerCase().trim();
  const result = new Set<string>();
  for (let i = 0; i < s.length - 1; i++) {
    result.add(s.slice(i, i + 2));
  }
  return result;
}

export function diceSimilarity(a: string, b: string): number {
  const cleanA = a.toLowerCase().trim();
  const cleanB = b.toLowerCase().trim();
  if (cleanA === cleanB) return 1;
  if (cleanA.length < 2 || cleanB.length < 2) return 0;

  const bgA = bigrams(cleanA);
  const bgB = bigrams(cleanB);
  let intersection = 0;
  for (const gram of bgA) {
    if (bgB.has(gram)) intersection += 1;
  }
  return (2 * intersection) / (bgA.size + bgB.size);
}

/**
 * Pipeline Step 4: Match against faculty member's previously used locations (fuzzy match >= 0.8).
 */
export function extractLocation(
  normalized: string,
  knownLocations: string[]
): { location?: string; matchedPhrase?: string } {
  const sortedLocations = [...knownLocations]
    .filter((l) => l.trim().length > 0)
    .sort((a, b) => b.length - a.length);

  for (const loc of sortedLocations) {
    const normLoc = loc.toLowerCase().trim();
    if (normalized.includes(normLoc)) {
      return { location: loc, matchedPhrase: normLoc };
    }

    const locWords = normLoc.split(/\s+/);
    const queryWords = normalized.split(/\s+/);
    const windowSize = locWords.length;

    if (queryWords.length >= windowSize) {
      for (let i = 0; i <= queryWords.length - windowSize; i++) {
        const candidate = queryWords.slice(i, i + windowSize).join(" ");
        if (diceSimilarity(candidate, normLoc) >= 0.8) {
          return { location: loc, matchedPhrase: candidate };
        }
      }
    }
  }

  return {};
}

/**
 * Pipeline Step 5: Remove stopwords and matched date, type, and location phrases; keep the rest.
 */
export function extractKeywords(
  normalized: string,
  phrasesToRemove: string[]
): string[] {
  let working = ` ${normalized} `;
  for (const phrase of phrasesToRemove) {
    if (!phrase) continue;
    const escaped = phrase.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    working = working.replace(new RegExp(`\\b${escaped}\\b`, "gi"), " ");
  }

  const tokens = working
    .trim()
    .split(/\s+/)
    .map((t) => t.replace(/^\.+|\.+$/g, ""))
    .filter((t) => t.length >= 2 && !STOPWORDS.has(t));

  return Array.from(new Set(tokens));
}

/**
 * Pipeline Step 6: Orchestrate steps 1–5 to build a structured filter.
 */
export function parseSearchQueryRuleBased(
  rawMessage: string,
  knownLocations: string[],
  referenceNow: Date = new Date()
): StructuredAiFilter {
  const normalized = normalizeQuery(rawMessage);
  if (!normalized) {
    return { keywords: [] };
  }

  const typeResult = detectEventType(normalized);
  const dateResult = extractDateRange(normalized, referenceNow);
  const locationResult = extractLocation(normalized, knownLocations);

  const phrasesToStrip = [
    ...typeResult.matchedPhrases,
    ...dateResult.matchedPhrases,
    ...(locationResult.matchedPhrase ? [locationResult.matchedPhrase] : []),
  ];

  const keywords = extractKeywords(normalized, phrasesToStrip);

  return {
    type: typeResult.type,
    from: dateResult.from,
    to: dateResult.to,
    dateLabel: dateResult.label,
    location: locationResult.location,
    keywords,
  };
}

import { Event } from "@prisma/client";
import { StructuredAiFilter } from "@/lib/validation/ai";

export interface ScoredEventResult {
  event: Event;
  score: number;
  matchReasons: string[];
}

/**
 * Pipeline Step 8: Score and rank candidate events.
 * - Date falls inside range: +40
 * - Each keyword found in title: +15; in description, organizer, or location: +5
 * - Type matches: +10
 * - Location matches: +15
 * - Results with score 0 are excluded. Return top 5.
 */
export function scoreAndRankEvents(
  events: Event[],
  filter: StructuredAiFilter
): ScoredEventResult[] {
  const scored: ScoredEventResult[] = [];

  for (const ev of events) {
    let score = 0;
    const reasons: string[] = [];

    if (filter.from && filter.to) {
      if (ev.startAt >= filter.from && ev.startAt <= filter.to) {
        score += 40;
        reasons.push(filter.dateLabel || "date range");
      }
    }

    if (filter.type && ev.type === filter.type) {
      score += 10;
      reasons.push(filter.type.toLowerCase());
    }

    if (
      filter.location &&
      ev.location &&
      ev.location.toLowerCase().includes(filter.location.toLowerCase())
    ) {
      score += 15;
      reasons.push(filter.location);
    }

    const titleLower = ev.title.toLowerCase();
    const secondaryBlob = [ev.description, ev.organizer, ev.location]
      .filter(Boolean)
      .join(" ")
      .toLowerCase();

    for (const kw of filter.keywords) {
      const kwLower = kw.toLowerCase();
      let matchedKw = false;

      if (titleLower.includes(kwLower)) {
        score += 15;
        matchedKw = true;
      }
      if (secondaryBlob.includes(kwLower)) {
        score += 5;
        matchedKw = true;
      }
      if (matchedKw) {
        reasons.push(kw);
      }
    }

    if (score > 0) {
      scored.push({
        event: ev,
        score,
        matchReasons: Array.from(new Set(reasons)),
      });
    }
  }

  return scored
    .sort((a, b) => {
      if (b.score !== a.score) return b.score - a.score;
      return b.event.startAt.getTime() - a.event.startAt.getTime();
    })
    .slice(0, 5);
}

export async function POST(req: NextRequest) {
  const startMs = Date.now();
  const path = "/api/auth/register";

  try {
    const body = await req.json();
    const parsed = registerSchema.parse(body);

    const existing = await prisma.faculty.findFirst({
      where: {
        OR: [{ facultyCode: parsed.facultyCode }, { email: parsed.email }],

import { NextRequest, NextResponse } from "next/server";
import { ZodError } from "zod";
import { prisma } from "@/lib/db";
import { authProvider } from "@/lib/auth/provider";
import { consumeAiRateLimit } from "@/lib/auth/rate-limit";
import { aiSearchRequestSchema } from "@/lib/validation/ai";
import { parseSearchQueryRuleBased } from "@/lib/ai/parse";
import { parseQueryWithLlm, isLlmModeEnabled } from "@/lib/ai/llm-adapter";
import { scoreAndRankEvents } from "@/lib/ai/score";
import { apiError, handleZodError, logRequest } from "@/lib/api-response";

const EXAMPLE_PROMPTS = [
  "that AI workshop last spring",
  "something in the seminar hall in March 2026",
  "events I hosted last month",
  "busy days this semester",
];

export async function POST(req: NextRequest) {
  const startMs = Date.now();
  const path = "/api/ai/search";

  const session = await authProvider.getSession();
  if (!session) {
    logRequest(path, 401, startMs);
    return apiError("UNAUTHORIZED", "Authentication required.", 401);
  }

  const rateLimit = consumeAiRateLimit(session.facultyId);
  if (!rateLimit.allowed) {
    logRequest(path, 429, startMs);
    return apiError(
      "RATE_LIMITED",
      "Rate limit reached (30 searches per hour). Please try again later.",
      429
    );
  }

  try {
    const body = await req.json();
    const { message } = aiSearchRequestSchema.parse(body);

    const locationRows = await prisma.event.findMany({
      where: {
        facultyId: session.facultyId,
        location: { not: null },
      },
      select: { location: true },
      distinct: ["location"],
    });
    const knownLocations = locationRows
      .map((r) => r.location)
      .filter((l): l is string => Boolean(l));

    let understood = isLlmModeEnabled()
      ? await parseQueryWithLlm(message)
      : null;

    if (!understood) {
      understood = parseSearchQueryRuleBased(message, knownLocations);
    }

    const hasAnyClue = Boolean(
      understood.type ||
        understood.from ||
        understood.to ||
        understood.location ||
        understood.keywords.length > 0
    );

    if (!hasAnyClue) {
      logRequest(path, 200, startMs);
      return NextResponse.json({
        understood,
        results: [],
        needsConfirmation: false,
        suggestions: EXAMPLE_PROMPTS,
        message:
          "I couldn't find specific keywords, dates, or places in your message. Try queries like: 'that AI workshop last spring' or 'events I hosted last month'.",
      });
    }

    const candidateEvents = await prisma.event.findMany({
      where: { facultyId: session.facultyId },
      orderBy: { startAt: "desc" },
    });

    const ranked = scoreAndRankEvents(candidateEvents, understood);
    const results = ranked.map((r) => ({
      ...r.event,
      score: r.score,
      matchReasons: r.matchReasons,
    }));

    if (results.length === 0) {
      logRequest(path, 200, startMs);
      return NextResponse.json({
        understood,
        results: [],
        needsConfirmation: false,
        suggestions: [
          "Try removing the date filter",
          "Search using a single keyword from the title",
        ],
        message:
          "No matching events were found in your records. Try broadening your search by removing the date or using a single keyword.",
      });
    }

    const needsConfirmation = results.length > 1;
    const replyMessage = needsConfirmation
      ? `I found ${results.length} matching events in your records. Is one of these the event you were looking for?`
      : `I found 1 matching event in your records.`;

    logRequest(path, 200, startMs);
    return NextResponse.json({
      understood,
      results,
      needsConfirmation,
      message: replyMessage,
    });
  } catch (err) {
    if (err instanceof ZodError) {
      logRequest(path, 400, startMs);
      return handleZodError(err);
    }
    logRequest(path, 500, startMs);
    return apiError("INTERNAL_ERROR", "Search assistant failed.", 500);
  }
}

import { NextRequest, NextResponse } from "next/server";
import { APP_CONFIG } from "@/config/constants";
import { verifySessionToken } from "@/lib/auth/jwt";

const PROTECTED_PREFIXES = [
  "/dashboard",
  "/events",
  "/calendar",
  "/achievements",
  "/reports",
  "/assistant",
  "/profile",
];

/**
 * F1.9: Protected routes redirect to /login with a `next` parameter when
 * no valid session exists.
 */
export async function middleware(req: NextRequest) {
  const { pathname, search } = req.nextUrl;
  const isProtected = PROTECTED_PREFIXES.some(
    (prefix) => pathname === prefix || pathname.startsWith(`${prefix}/`)
  );

  if (!isProtected) {
    return NextResponse.next();
  }

  const token = req.cookies.get(APP_CONFIG.sessionCookieName)?.value;
  const session = token ? await verifySessionToken(token) : null;

  if (!session) {
    const loginUrl = new URL("/login", req.url);
    loginUrl.searchParams.set("next", `${pathname}${search}`);
    return NextResponse.redirect(loginUrl);
  }

  return NextResponse.next();
}
